  --episodes EPISODES
  --attack {clean,bias,ramp,random_bias,random_ramp}
  --epsilon EPSILON
  --duration DURATION
  --monitor-threshold MONITOR_THRESHOLD
                        Frozen calibrated threshold in metres; enables
                        correction
  --legacy-sampling
